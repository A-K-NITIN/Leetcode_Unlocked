select p.product_id ,round(sum(p.price*(case when u.units is null then 0 else u.units end))/sum(case when u.units is null or u.units = 0 then 1 else u.units end),2)as average_price from Prices as p left join UnitsSold as u on (p.product_id=u.product_id) where (u.purchase_date between p.start_date and p.end_date) or u.purchase_date is null group by p.product_id;
